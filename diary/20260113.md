### Model

```Swift
import SwiftUI
import Combine
import AVFoundation

class AudioEnginePlayerModel: NSObject, ObservableObject {
    let audioPlayer = AVAudioEngine()
    private var playerNode: AVAudioPlayerNode?
    private var delayNode: AVAudioUnitDelay?

    // アセットカタログに追加したアセット名の定数化
    private let audioAssetName = "sound"

    func playAudio() {
        guard let dataAsset = loadAudioAsset(name: audioAssetName) else {
            return
        }

        do {
            // 複雑になってしまったコードは、関数にして実行する。責任を分ける
            resetEngine()
            let (audioFile, tempURL) = try createAudioFile(from: dataAsset)
            setupAudioNodes()
            try scheduleAndPlay(file: audioFile, tempURL: tempURL)
        } catch {
            print("Failed to create the audio player: \(error)")
        }
    }

    // MARK: - Private Helper Methods

    /// アセットカタログから音声ファイルを読み込む
    private func loadAudioAsset(name: String) -> NSDataAsset? {
        guard let dataAsset = NSDataAsset(name: name) else {
            print("Failed to find the audio file in assets: \(name)")
            return nil
        }
        return dataAsset
    }

    /// エンジンを停止してリセットする
    private func resetEngine() {
        guard audioPlayer.isRunning else { return }

        audioPlayer.stop()

        if let existingPlayerNode = playerNode {
            existingPlayerNode.stop()
            audioPlayer.detach(existingPlayerNode)
        }

        if let existingDelayNode = delayNode {
            audioPlayer.detach(existingDelayNode)
        }
    }

    /// データアセットから一時ファイルを作成し、AVAudioFileを生成する
    private func createAudioFile(from dataAsset: NSDataAsset) throws -> (AVAudioFile, URL) {
        let tempURL = FileManager.default.temporaryDirectory
            .appendingPathComponent(UUID().uuidString)
            .appendingPathExtension("wav")

        try dataAsset.data.write(to: tempURL)
        let audioFile = try AVAudioFile(forReading: tempURL)

        return (audioFile, tempURL)
    }

    /// オーディオノードを作成して接続する
    private func setupAudioNodes() {
        // Playerノードの作成と接続
        let newPlayerNode = AVAudioPlayerNode()
        audioPlayer.attach(newPlayerNode)
        playerNode = newPlayerNode

        // Delayノードの作成と接続
        let newDelayNode = AVAudioUnitDelay()
        newDelayNode.wetDryMix = 1.0
        newDelayNode.delayTime = 1.0
        audioPlayer.attach(newDelayNode)
        delayNode = newDelayNode

        // ノードを接続: Player -> Delay -> MainMixer
        audioPlayer.connect(newPlayerNode, to: newDelayNode, format: nil)
        audioPlayer.connect(newDelayNode, to: audioPlayer.mainMixerNode, format: nil)
    }

    /// ファイルをスケジュールして再生する
    private func scheduleAndPlay(file: AVAudioFile, tempURL: URL) throws {
        guard let playerNode = playerNode else {
            throw NSError(domain: "AudioEnginePlayerModel", code: 1, userInfo: [NSLocalizedDescriptionKey: "Player node is not initialized"])
        }

        // ファイルをスケジュール（再生キューに追加）
        playerNode.scheduleFile(file, at: nil) { [weak self] in
            // 再生完了後に一時ファイルを削除
            DispatchQueue.main.asyncAfter(deadline: .now() + 1.0) {
                try? FileManager.default.removeItem(at: tempURL)
            }
        }

        // エンジンを開始して再生
        try audioPlayer.start()
        playerNode.play()
    }
}
```
